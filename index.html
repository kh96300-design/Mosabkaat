import { useEffect, useRef } from 'react'; interface ConfettiCanvasProps { triggerKey: number | string | null; } export function ConfettiCanvas({ triggerKey }: ConfettiCanvasProps) { const canvasRef = useRef(null); const prevKeyRef = useRef(triggerKey); useEffect(() => { if (triggerKey === null || triggerKey === prevKeyRef.current) { prevKeyRef.current = triggerKey; return; } prevKeyRef.current = triggerKey; if (typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches) { return; } const canvas = canvasRef.current; if (!canvas) return; const ctx = canvas.getContext('2d'); if (!ctx) return; canvas.width = window.innerWidth; canvas.height = window.innerHeight; const palette = ['#3b5bfd', '#10b981', '#f59e0b', '#e11d48']; const pieces = Array.from({ length: 90 }, () => ({ x: Math.random() * canvas.width, y: Math.random() * -canvas.height, r: Math.random() * 6 + 4, c: palette[Math.floor(Math.random() * palette.length)], v: Math.random() * 4 + 2, })); let frames = 0; let rafId = 0; const draw = () => { ctx.clearRect(0, 0, canvas.width, canvas.height); for (const p of pieces) { p.y += p.v; ctx.fillStyle = p.c; ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2); ctx.fill(); } if (++frames < 140) { rafId = requestAnimationFrame(draw); } else { ctx.clearRect(0, 0, canvas.width, canvas.height); } }; rafId = requestAnimationFrame(draw); return () => { cancelAnimationFrame(rafId); ctx.clearRect(0, 0, canvas.width, canvas.height); }; }, [triggerKey]); return ; }

const STATE_PROPERTY_KEY = "CONTEST_SYNC_STATE_V1"; const LAST_LOGS_SIG_KEY = "CONTEST_LAST_LOGS_SIG_V1"; const SHEET_LOGS_NAME = "سجل_الأحداث"; const SHEET_RESULTS_NAME = "النتائج_والمتسابقين"; const SHEET_FOLLOWERS_NAME = "المتابعون_والمشاهدون"; function doGet(e) { try { const action = (e && e.parameter && e.parameter.action) ? String(e.parameter.action) : "get_state"; const lock = LockService.getScriptLock(); lock.waitLock(5000); const rawState = PropertiesService.getScriptProperties().getProperty(STATE_PROPERTY_KEY); lock.releaseLock(); let parsedState = null; if (rawState) { try { parsedState = JSON.parse(rawState); } catch (err) { parsedState = null; } } return ContentService .createTextOutput(JSON.stringify({ ok: true, action: action, state: parsedState, serverTime: Date.now() })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: error.message })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return jsonResponse({ ok: false, error: "Empty payload" }); } const body = JSON.parse(e.postData.contents); const incomingState = body.state; const reason = body.reason ? String(body.reason) : "تحديث"; if (!incomingState || typeof incomingState !== "object") { return jsonResponse({ ok: false, error: "Invalid state object" }); } const lock = LockService.getScriptLock(); lock.waitLock(5000); const props = PropertiesService.getScriptProperties(); const existingRaw = props.getProperty(STATE_PROPERTY_KEY); let shouldSaveState = true; if (existingRaw) { try { const existingState = JSON.parse(existingRaw); const oldTs = Number(existingState.lastUpdated || 0); const newTs = Number(incomingState.lastUpdated || Date.now()); if (newTs < oldTs) { shouldSaveState = false; } } catch (err) { shouldSaveState = true; } } if (!incomingState.lastUpdated) { incomingState.lastUpdated = Date.now(); } if (shouldSaveState) { props.setProperty(STATE_PROPERTY_KEY, JSON.stringify(incomingState)); syncStateToSheets(incomingState, reason, props); } lock.releaseLock(); return jsonResponse({ ok: true, saved: shouldSaveState, lastUpdated: incomingState.lastUpdated }); } catch (error) { return jsonResponse({ ok: false, error: error.message }); } } function syncStateToSheets(stateObj, reason, props) { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) return; const updatedAtStr = Utilities.formatDate( new Date(Number(stateObj.lastUpdated || Date.now())), Session.getScriptTimeZone() || "Asia/Muscat", "yyyy-MM-dd HH:mm:ss" ); writeLogsSheet(ss, stateObj, reason, updatedAtStr, props); writeResultsSheet(ss, stateObj, updatedAtStr); writeFollowersSheet(ss, stateObj, reason, updatedAtStr); } function getOrCreateSheet(ss, name, headers) { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.setRightToLeft(true); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } function writeLogsSheet(ss, stateObj, reason, updatedAtStr, props) { const sheet = getOrCreateSheet(ss, SHEET_LOGS_NAME, [ "وقت التسجيل في الشيت", "سبب المزامنة", "نص الحدث" ]); const logs = Array.isArray(stateObj.logs) ? stateObj.logs : []; if (logs.length === 0) { props.setProperty(LAST_LOGS_SIG_KEY, ""); return; } // إذا بدأ سجل جديد بعد تصفير المسابقة، نعيد ضبط البصمة const prevTop = props.getProperty(LAST_LOGS_SIG_KEY) || ""; const currentTop = String(logs[0] || ""); // نجمع الأحداث الجديدة التي لم تُسجل بعد بناءً على آخر حدث مسجل let newEntries = []; if (!prevTop) { newEntries = logs.slice().reverse(); } else { const idxInLogs = logs.indexOf(prevTop); if (idxInLogs === -1) { // في حال تم تصفير السجل أو بدء مسابقة جديدة newEntries = logs.slice().reverse(); } else if (idxInLogs > 0) { newEntries = logs.slice(0, idxInLogs).reverse(); } } if (newEntries.length > 0) { const rows = newEntries.map(entry => [updatedAtStr, reason, String(entry)]); sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 3).setValues(rows); } props.setProperty(LAST_LOGS_SIG_KEY, currentTop); } function writeResultsSheet(ss, stateObj, updatedAtStr) { const sheet = getOrCreateSheet(ss, SHEET_RESULTS_NAME, [ "الرقم", "اسم المتسابق", "الحالة", "الإسقاطات", "الأخطاء", "الاعتذارات", "فائز؟", "آخر تحديث" ]); const players = Array.isArray(stateObj.players) ? stateObj.players : []; const winners = Array.isArray(stateObj.winners) ? stateObj.winners : []; const winnerIds = new Set(winners.map(w => Number(w.id))); // إعادة كتابة جدول النتائج المحدّث بشكل نظيف دون تكرار صفوف اللاعبين const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, 8).clearContent(); } if (players.length > 0) { const rows = players.map(p => { let statusAr = "متاح"; if (p.status === "eliminated") statusAr = "مستبعد / منسحب"; else if (p.status === "winner") statusAr = "فائز"; const isWinner = winnerIds.has(Number(p.id)) || p.status === "winner" ? "نعم 🏆" : "لا"; return [ Number(p.id || 0), String(p.name || ("متسابق #" + p.id)), statusAr, Number(p.kills || 0), Number(p.errors || 0), Number(p.passes || 0), isWinner, updatedAtStr ]; }); sheet.getRange(2, 1, rows.length, 8).setValues(rows); } } function writeFollowersSheet(ss, stateObj, reason, updatedAtStr) { const sheet = getOrCreateSheet(ss, SHEET_FOLLOWERS_NAME, [ "وقت التحديث", "إجمالي المتابعين", "عدد الشاشات المرتبة", "عدد الفائزين الحالي", "ملاحظة / سبب التحديث" ]); const followersCount = Number(stateObj.followersCount || 0); const screensCount = Number(stateObj.orderedScreensCount || 1); const winnersCount = Array.isArray(stateObj.winners) ? stateObj.winners.length : 0; // نضيف صفاً جديداً عندما يتغير عدد المتابعين أو عند تحديثات رئيسية const lastRow = sheet.getLastRow(); let shouldAppend = true; if (lastRow > 1) { const prevFollowers = Number(sheet.getRange(lastRow, 2).getValue() || 0); const prevReason = String(sheet.getRange(lastRow, 5).getValue() || ""); if (prevFollowers === followersCount && prevReason === reason) { shouldAppend = false; } } if (shouldAppend) { sheet.appendRow([updatedAtStr, followersCount, screensCount, winnersCount, reason]); } } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
